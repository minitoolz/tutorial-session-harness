% style: paper — academic look: compact headings, running header, numbered eqs by section
\usepackage{mathtools}
% Bold math. Pandoc's XeLaTeX template loads unicode-math, which breaks bm's \bm and
% amsmath's \boldsymbol on Greek letters ("Improper alphabetic constant") and makes
% \mathbf drop Greek glyphs. Route all three through unicode-math's math-font alphabets.
\makeatletter
\@ifpackageloaded{unicode-math}{%
  \unimathsetup{mathbf=sym}%
  \DeclareRobustCommand{\bm}[1]{\symbfit{#1}}%
  \DeclareRobustCommand{\boldsymbol}[1]{\symbfit{#1}}%
}{\usepackage{bm}}
\makeatother
\usepackage{float}
\floatplacement{figure}{htbp}
\usepackage[font=small,labelfont=bf]{caption}
\usepackage{booktabs}
\usepackage{titlesec}
\titleformat*{\section}{\large\bfseries}
\titleformat*{\subsection}{\normalsize\bfseries}
\titleformat*{\subsubsection}{\normalsize\itshape}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\makeatletter
\AtBeginDocument{\let\tutortitle\@title}% \maketitle clears \@title; keep a copy
\makeatother
\fancyhead[L]{\small\itshape\tutortitle}
\fancyhead[R]{\small\thepage}
\renewcommand{\headrulewidth}{0.3pt}
\numberwithin{equation}{section}
